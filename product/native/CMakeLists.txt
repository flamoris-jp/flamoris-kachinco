cmake_minimum_required(VERSION 3.20)
project(KachincoNative VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Native runtime supports Windows x64 and Linux/glibc test hosts")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Kachinco native runtime requires a 64-bit target")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(kachinco_native SHARED src/runtime.cpp src/process.cpp src/cache.cpp src/decoded_media.cpp src/compositor.cpp src/timeline.cpp src/playback.cpp src/editor_domain.cpp src/project_codec.cpp src/editor_session.cpp)
target_include_directories(kachinco_native PUBLIC include)
target_compile_definitions(kachinco_native PRIVATE KN_BUILD)
set_target_properties(kachinco_native PROPERTIES OUTPUT_NAME "Kachinco.Native.Runtime" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(MSVC)
  target_compile_options(kachinco_native PRIVATE /W4 /WX /EHsc /fp:strict)
else()
  target_compile_options(kachinco_native PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
find_package(Threads REQUIRED)
target_link_libraries(kachinco_native PRIVATE Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(native_contracts tests/contracts.cpp tests/c_header.c)
  target_link_libraries(native_contracts PRIVATE kachinco_native)
  add_test(NAME native_contracts COMMAND native_contracts)
  add_executable(native_process_contracts tests/process_contracts.cpp)
  target_link_libraries(native_process_contracts PRIVATE kachinco_native Threads::Threads)
  add_test(NAME native_process_contracts COMMAND native_process_contracts)
  add_executable(native_render_contracts tests/render_contracts.cpp)
  target_link_libraries(native_render_contracts PRIVATE kachinco_native)
  add_test(NAME native_render_contracts COMMAND native_render_contracts)
  add_executable(native_editor_contracts tests/editor_contracts.cpp)
  target_link_libraries(native_editor_contracts PRIVATE kachinco_native)
  add_test(NAME native_editor_contracts COMMAND native_editor_contracts)
  set_tests_properties(native_process_contracts PROPERTIES TIMEOUT 45)
endif()
